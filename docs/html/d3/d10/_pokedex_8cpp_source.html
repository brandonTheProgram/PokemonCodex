125Json::Value Pokedex::pokemonButtonDataToJsonValue(
126 const std::vector<std::vector<std::string>>& results)
127{
128 Json::Value pokemonButtons(Json::arrayValue);
129
130 this->logger_.debug("Pokedex::pokemonButtonDataToJsonValue invoked");
131
132 try
133 {
134 for (const auto& row : results)
135 {
136 Json::Value pokemon(Json::objectValue);
137
138 pokemon["pokedex_number"] = row.at(0);
139 pokemon["region_id"] = row.at(1);
140 pokemon["name"] = row.at(2);
141 pokemon["image"] = row.at(3);
142
143 // Apply the region form to the name if present
144 std::string regionId = pokemon["region_id"].asString();
145 if (regionId != "NULL")
146 {
147 // Append the name of the regional form to the Pokemon's name
148 pokemon["name"] = Json::Value(
149 this->regionRepository_->queryRegionalFormTable(std::stoi(regionId)) + " " +
150 pokemon["name"].asString());
151 }
152
153 logger_.debug("Pokedex::pokemonButtonDataToJsonValue Found: " +
154 pokemon["name"].asString());
155
156 pokemonButtons.append(pokemon);
157 }
158 }
159 catch (const std::invalid_argument& e)
160 {
161 this->logger_.critical("Pokedex::pokemonButtonDataToJsonValue caught an exception: " +
162 std::string(e.what()));
163 return Json::Value(Json::arrayValue);
164 }
165
166 return pokemonButtons;
167}

200void Pokedex::connectEvolutionaryChain(std::vector<EvolutionData>& evolutionaryChain) const
201{
202 this->logger_.debug("Pokedex::connectEvolutionaryChain invoked");
203
204 // Map to store depth values for sorting
205 std::unordered_map<std::pair<std::uint32_t, std::uint32_t>, int, pair_hash> depthMap;
206
207 // Define assignDepth as a std::function to allow recursion
208 std::function<void(const EvolutionData&, int)> assignDepth =
209 [&](const EvolutionData& data, int depth)
210 {
211 auto key = std::make_pair(data.basePokedexNumber, data.baseRegionId);
212 if (depthMap.find(key) == depthMap.end())
213 {
214 depthMap[key] = depth;
215
216 // Look for evolutions starting from this Pokémon
217 for (const auto& evolution : evolutionaryChain)
218 {
219 if (evolution.basePokedexNumber == data.evolvedPokedexNumber &&
220 evolution.baseRegionId == data.evolvedRegionId)
221 {
222 assignDepth(evolution, depth + 1);
223 }
224 }
225 }
226 };
227
228 // Start assigning depths from Pokémon with no predecessors
229 for (const auto& data : evolutionaryChain)
230 {
231 auto key = std::make_pair(data.basePokedexNumber, data.baseRegionId);
232 if (std::none_of(evolutionaryChain.begin(), evolutionaryChain.end(),
233 [&](const EvolutionData& other)
234 {
235 return other.evolvedPokedexNumber == data.basePokedexNumber &&
236 other.evolvedRegionId == data.baseRegionId;
237 }))
238 {
239 assignDepth(data, 0); // Start from depth 0
240 }
241 }
242
243 // Sort the evolutionary chain by depth, prioritizing pre-evolutions and regional splits
244 std::sort(evolutionaryChain.begin(), evolutionaryChain.end(),
245 [&](const EvolutionData& a, const EvolutionData& b)
246 {
247 auto keyA = std::make_pair(a.basePokedexNumber, a.baseRegionId);
248 auto keyB = std::make_pair(b.basePokedexNumber, b.baseRegionId);
249
250 // Primary sorting by depth
251 if (depthMap[keyA] != depthMap[keyB]) return depthMap[keyA] < depthMap[keyB];
252
253 // Secondary sorting: prioritize non-regional forms over regional forms
254 if (a.baseRegionId != b.baseRegionId) return a.baseRegionId < b.baseRegionId;
255
256 // Tertiary sorting: prioritize evolved regional forms consistently
257 if (a.evolvedRegionId != b.evolvedRegionId)
258 return a.evolvedRegionId < b.evolvedRegionId;
259
260 // Final fallback: order by Pokédex numbers
261 return a.evolvedPokedexNumber < b.evolvedPokedexNumber;
262 });
263}
